**Goal:** Turn Kira's two-demo tool registry into a harness a user can trust — real workspace tools, per-tool `allow/ask/deny` with live approvals, file checkpoints with diff and restore, declarative HTTP plugins, and the two spend-accounting fixes from sub-project 9's findings.

**Architecture:** The engine gains five focused modules (`limits`, `workspace`, `policy`, `approvals`, `checkpoints`, `plugins`) around the existing registry and loop. The client gains a Tools route, an approval card, and six live work-panel tabs.

**Tech Stack:** Python 3.12 + FastAPI + httpx + sqlite3 (engine), React 19 + TypeScript + Vitest + jsdom (client), hand-written CSS.

```python
from __future__ import annotations

import json
from pathlib import Path

from .limits import (
    MAX_LIST_ENTRIES,
    MAX_READ_BYTES,
    MAX_SEARCH_BYTES,
    MAX_SEARCH_FILES,
    MAX_SEARCH_HITS,
)


class WorkspaceError(ValueError):
    pass


class Workspace:
    """Every tool path resolves here. Symlinks are followed *before* the containment check."""

    def __init__(self, root: Path) -> None:
        self.root = Path(root).resolve()
        self.root.mkdir(parents=True, exist_ok=True)

    def safe_path(self, raw: str) -> Path:
        if not isinstance(raw, str) or not raw or "\x00" in raw:
            raise WorkspaceError("path must be a non-empty string")
        candidate = Path(raw)
        resolved = (candidate if candidate.is_absolute() else self.root / candidate).resolve()
        if resolved != self.root and self.root not in resolved.parents:
            raise WorkspaceError(f"path escapes the workspace: {raw}")
        return resolved

    def read_text(self, raw: str) -> str:
        path = self.safe_path(raw)
        if not path.exists():
            raise WorkspaceError(f"file does not exist: {raw}")
        if not path.is_file():
            raise WorkspaceError(f"not a file: {raw}")
        data = path.read_bytes()
        if b"\x00" in data:
            raise WorkspaceError(f"file looks binary: {raw}")
        text = data[:MAX_READ_BYTES].decode("utf-8", "replace")
        if len(data) > MAX_READ_BYTES:
            text += f"\n[truncated at {MAX_READ_BYTES} bytes]"
        return text

    def list_dir(self, raw: str) -> str:
        path = self.safe_path(raw)
        if not path.is_dir():
            raise WorkspaceError(f"not a directory: {raw}")
        entries = []
        for child in sorted(path.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower()))[:MAX_LIST_ENTRIES]:
            try:
                size = child.stat().st_size if child.is_file() else 0
            except OSError:
                size = 0
            entries.append({"name": child.name, "type": "dir" if child.is_dir() else "file", "size": size})
        return json.dumps({"path": raw, "entries": entries, "count": len(entries)}, ensure_ascii=False)

    def search_text(self, query: str, raw: str) -> str:
        if not query:
            raise WorkspaceError("query must not be empty")
        root = self.safe_path(raw)
        if not root.is_dir():
            raise WorkspaceError(f"not a directory: {raw}")
        hits: list[dict] = []
        scanned = 0
        budget = MAX_SEARCH_BYTES
        for path in sorted(root.rglob("*")):
            if len(hits) >= MAX_SEARCH_HITS or scanned >= MAX_SEARCH_FILES or budget <= 0:
                break
            if not path.is_file() or ".git" in path.parts:
                continue
            scanned += 1
            try:
                data = path.read_bytes()[: min(budget, MAX_READ_BYTES)]
            except OSError:
                continue
            budget -= len(data)
            if b"\x00" in data:
                continue
            for number, line in enumerate(data.decode("utf-8", "replace").splitlines(), start=1):
                if query in line:
                    hits.append({"path": path.relative_to(self.root).as_posix(), "line": number, "text": line[:300]})
                    if len(hits) >= MAX_SEARCH_HITS:
                        break
        return json.dumps({"query": query, "hits": hits, "scanned_files": scanned}, ensure_ascii=False)
```

```python
    def write_text(self, raw: str, content: str) -> int:
        if not isinstance(content, str):
            raise WorkspaceError("content must be a string")
        data = content.encode("utf-8")
        if len(data) > MAX_WRITE_BYTES:
            raise WorkspaceError(f"content is too large (max {MAX_WRITE_BYTES} bytes)")
        path = self.safe_path(raw)
        if path == self.root:
            raise WorkspaceError("cannot write the workspace root")
        path.parent.mkdir(parents=True, exist_ok=True)
        temporary = path.with_name(path.name + ".kira-tmp")
        temporary.write_bytes(data)
        os.replace(temporary, path)
        return len(data)

    def _command_env(self) -> dict[str, str]:
        env = {key: value for key in ENV_ALLOW if (value := os.environ.get(key)) is not None}
        # Windows needs these even with a stripped environment; without SYSTEMROOT some runtimes fail to start.
        env.setdefault("PATH", os.defpath)
        return env

    async def run_command(self, argv: list[str], cwd: str = ".", timeout_s: int = 30) -> dict:
        if not isinstance(argv, list) or not argv or len(argv) > MAX_ARGV:
            raise WorkspaceError(f"argv must be a list of 1-{MAX_ARGV} strings")
        if not all(isinstance(item, str) and item and len(item) <= MAX_ARG_CHARS for item in argv):
            raise WorkspaceError("every argv entry must be a non-empty string under 4096 characters")
        name = Path(argv[0]).name.lower()
        if name in SHELLS:
            raise WorkspaceError(f"refusing to run a shell ({name}); pass the program directly")
        if not (MIN_TIMEOUT_S <= timeout_s <= MAX_TIMEOUT_S):
            raise WorkspaceError(f"timeout must be {MIN_TIMEOUT_S}-{MAX_TIMEOUT_S} seconds")
        resolved_cwd = self.safe_path(cwd)
        if not resolved_cwd.is_dir():
            raise WorkspaceError(f"not a directory: {cwd}")
        executable = argv[0]
        if not Path(executable).is_absolute() and "/" not in executable and "\\" not in executable:
            executable = shutil.which(executable, path=self._command_env().get("PATH")) or executable
        started = time.perf_counter()
        process = await asyncio.create_subprocess_exec(
            executable,
            *argv[1:],
            cwd=str(resolved_cwd),
            env=self._command_env(),
            stdin=asyncio.subprocess.DEVNULL,
            stdout=asyncio.subprocess.PIPE,
            stderr=asyncio.subprocess.PIPE,
        )
        timed_out = False
        try:
            stdout, stderr = await asyncio.wait_for(process.communicate(), timeout=timeout_s)
        except asyncio.TimeoutError:
            timed_out = True
            process.kill()
            stdout, stderr = await process.communicate()
        duration_ms = int((time.perf_counter() - started) * 1000)
        out = stdout[:MAX_OUTPUT_BYTES].decode("utf-8", "replace")
        err = stderr[:MAX_OUTPUT_BYTES].decode("utf-8", "replace")
        truncated = len(stdout) > MAX_OUTPUT_BYTES or len(stderr) > MAX_OUTPUT_BYTES
        return {
            "exit_code": 124 if timed_out else int(process.returncode or 0),
            "stdout": out,
            "stderr": err,
            "duration_ms": duration_ms,
            "truncated": truncated,
        }
```

```python
        value = args.get(key, default)
        if not isinstance(value, str):
            raise WorkspaceError(f"{key} must be a string")
        return value

        return workspace.list_dir(path_of(args))

    def read_file(args: dict) -> str:
        return workspace.read_text(path_of(args, default=""))

    def search_text(args: dict) -> str:
        query = args.get("query")
        if not isinstance(query, str) or not query:
            raise WorkspaceError("query must be a non-empty string")
        return workspace.search_text(query, path_of(args))

    def write_file(args: dict) -> str:
        content = args.get("content")
        if not isinstance(content, str):
            raise WorkspaceError("content must be a string")
```

```python
    if tool.name in overrides:

    return tool.default_policy or RISK_DEFAULTS.get(tool.risk, "ask")
```

```python
    def expire_all(self) -> int:
        return self.store.expire_pending_approvals()

    def open(self, *, session_id: str | None, tool: str, arguments: dict, risk: str) -> dict:
        row = self.store.add_approval(session_id, tool, arguments, risk, expires_at=time.time() + self.timeout_s)
        self._waiters[row["id"]] = asyncio.get_running_loop().create_future()
        return row

    async def wait(self, approval_id: str) -> Decision:
        future = self._waiters.get(approval_id)
        if future is None:
            row = self.store.get_approval(approval_id)
            return Decision(False, "no live approval for this turn")
        try:
            return await asyncio.wait_for(future, timeout=self.timeout_s)
        except asyncio.TimeoutError:
            self.store.decide_approval(approval_id, "deny", reason="approval timed out")
            return Decision(False, "approval timed out")
        finally:
            self._waiters.pop(approval_id, None)

    def decide(self, approval_id: str, decision: str) -> dict | None:
        row = self.store.decide_approval(approval_id, decision)
        future = self._waiters.get(approval_id)
        if (future is not None and not future.done() and row is not None):
            future.set_result(Decision(row["status"] == "allowed", row["reason"]))
        return row

    def cancel(self, approval_id: str) -> None:
        row = self.store.decide_approval(approval_id, "deny", reason="the turn ended before a decision")
        future = self._waiters.get(approval_id)
        if future is not None and not future.done():
            future.set_result(Decision(False, "the turn ended before a decision"))
```

```python
from __future__ import annotations

import difflib
import hashlib
import json
import os
import shutil
from pathlib import Path

from .limits import MAX_CHECKPOINT_BYTES
from .store import Store
from .workspace import Workspace


class CheckpointConflict(Exception):
    def __init__(self, conflicts: list[dict]) -> None:
        super().__init__(f"{len(conflicts)} file(s) changed since the checkpoint")
        self.conflicts = conflicts


def _sha256(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


class CheckpointManager:
    def __init__(self, workspace: Workspace, directory: Path, store: Store, retention: int = 50) -> None:
        self.workspace = workspace
        self.directory = Path(directory)
        self.directory.mkdir(parents=True, exist_ok=True)
        self.store = store
        self.retention = retention

    def take(self, *, session_id: str | None, tool: str, call_id: str, paths: list[str]) -> dict:
        files = []
        for index, raw in enumerate(paths):
            resolved = self.workspace.safe_path(raw)
            relative = resolved.relative_to(self.workspace.root).as_posix()
            entry = {"path": relative, "existed": False, "sha256_before": None, "sha256_after": None, "size": 0, "blob": None}
            if resolved.exists():
                if not resolved.is_file():
                    raise ValueError(f"not a file: {raw}")
                data = resolved.read_bytes()
                if len(data) > MAX_CHECKPOINT_BYTES:
                    raise ValueError(f"file too large to checkpoint: {relative}")
                entry.update(existed=True, sha256_before=_sha256(data), size=len(data), blob=f"{index}.blob")
            files.append(entry)
        row = self.store.add_checkpoint(session_id=session_id, tool=tool, call_id=call_id, scope="files", files=files)
        target = self.directory / row["id"]
        target.mkdir(parents=True, exist_ok=True)
        for index, entry in enumerate(files):
            if entry["existed"]:
                resolved = self.workspace.root / entry["path"]
                (target / f"{index}.blob").write_bytes(resolved.read_bytes())
        (target / "manifest.json").write_text(json.dumps(row, indent=2), encoding="utf-8")
        return row

    def finalize(self, checkpoint_id: str) -> dict:
        row = self.store.get_checkpoint(checkpoint_id)
        if row is None:
            raise KeyError(checkpoint_id)
        files = []
        for entry in row["files"]:
            current = self.workspace.root / entry["path"]
            after = _sha256(current.read_bytes()) if current.is_file() else None
            files.append({**entry, "sha256_after": after})
        updated = self.store.finalize_checkpoint(checkpoint_id, files)
        self.prune()
        return updated

    def get(self, checkpoint_id: str) -> dict | None:
        return self.store.get_checkpoint(checkpoint_id)

    def list(self, limit: int = 20, session_id: str | None = None) -> list[dict]:
        return self.store.list_checkpoints(limit=limit, session_id=session_id)

    def prune(self) -> int:
        removed = self.store.prune_checkpoints(self.retention)
        for checkpoint_id in removed:
            shutil.rmtree(self.directory / checkpoint_id, ignore_errors=True)
        return len(removed)

    def _blob_path(self, checkpoint_id: str, entry: dict) -> Path:
        return self.directory / checkpoint_id / entry["blob"]

    def diff(self, checkpoint_id: str) -> dict:
        row = self.store.get_checkpoint(checkpoint_id)
        if row is None:
            raise KeyError(checkpoint_id)
        files = []
        for entry in row["files"]:
            current = self.workspace.root / entry["path"]
            before = self._blob_path(checkpoint_id, entry).read_bytes() if entry["existed"] else b""
            after = current.read_bytes() if current.is_file() else b""
            if b"\x00" in before or b"\x00" in after:
                body, changed = "binary differs", before != after
            else:
                old = before.decode("utf-8", "replace").splitlines(keepends=True)
                new = after.decode("utf-8", "replace").splitlines(keepends=True)
                patch = list(difflib.unified_diff(old, new, fromfile=f"a/{entry['path']}", tofile=f"b/{entry['path']}"))
                body = "".join(patch[:200])
                changed = bool(patch)
            files.append({"path": entry["path"], "existed": entry["existed"], "changed": changed, "diff": body})
        return {"checkpoint": row, "files": files}

    def restore(self, checkpoint_id: str, *, force: bool = False) -> dict:
        row = self.store.get_checkpoint(checkpoint_id)
        if row is None:
            raise KeyError(checkpoint_id)
        conflicts = []
        for entry in row["files"]:
            current = self.workspace.root / entry["path"]
            actual = _sha256(current.read_bytes()) if current.is_file() else None
            if actual != entry["sha256_after"]:
                conflicts.append({"path": entry["path"], "expected_after": entry["sha256_after"], "actual": actual})
        if conflicts and not force:
            raise CheckpointConflict(conflicts)
        safety = self.take(session_id=row["session_id"], tool="restore", call_id=row["id"], paths=[entry["path"] for entry in row["files"]])
        self.finalize(safety["id"])
        for entry in row["files"]:
            target = self.workspace.root / entry["path"]
            if entry["existed"]:
                data = self._blob_path(checkpoint_id, entry).read_bytes()
                target.parent.mkdir(parents=True, exist_ok=True)
                temporary = target.with_name(target.name + ".kira-tmp")
                temporary.write_bytes(data)
                os.replace(temporary, target)
            elif target.exists():
                target.unlink()
        self.store.mark_checkpoint_restored(checkpoint_id)
        return {"restored": [entry["path"] for entry in row["files"]], "conflicts": conflicts, "checkpoint_id": safety["id"]}
```

```python
from __future__ import annotations

import json
import re
from dataclasses import dataclass
from pathlib import Path
from urllib.parse import quote

import httpx

from .limits import MAX_PLUGIN_BYTES, MAX_PLUGIN_RESPONSE
from .tools import ToolDef, ToolOutcome
from .vault import Vault

NAME = re.compile(r"^[a-z][a-z0-9_]{2,39}$")
PLACEHOLDER = re.compile(r"\{([^{}]+)\}")
RISKS = ("read", "write", "exec", "network")
POLICIES = ("allow", "ask", "deny")


@dataclass
class PluginStatus:
    file: str
    name: str
    status: str
    reason: str = ""

    def to_wire(self) -> dict:
        return {"file": self.file, "name": self.name, "status": self.status, "reason": self.reason}


def _templates(value, path: str, params: set[str]) -> None:
    if isinstance(value, str):
        for name in PLACEHOLDER.findall(value):
            if name not in params:
                raise ValueError(f"{path} names an undeclared placeholder {{{name}}}")
    elif isinstance(value, dict):
        for key, item in value.items():
            _templates(item, f"{path}.{key}", params)
    elif isinstance(value, list):
        for index, item in enumerate(value):
            _templates(item, f"{path}[{index}]", params)


def _fill(template: str, values: dict) -> str:
    return PLACEHOLDER.sub(lambda match: quote(str(values[match.group(1)]), safe=""), template)


def _parse(path: Path) -> dict:
    if path.stat().st_size > MAX_PLUGIN_BYTES:
        raise ValueError(f"file is larger than {MAX_PLUGIN_BYTES} bytes")
    raw = json.loads(path.read_text(encoding="utf-8"))
    if not isinstance(raw, dict):
        raise ValueError("plugin file must be a JSON object")
    name = str(raw.get("name", ""))
    if not NAME.match(name):
        raise ValueError("name must match ^[a-z][a-z0-9_]{2,39}$")
    if not str(raw.get("description", "")).strip():
        raise ValueError("description must not be empty")
    risk = str(raw.get("risk", "network"))
    if risk not in RISKS:
        raise ValueError(f"risk must be one of {', '.join(RISKS)}")
    policy = raw.get("policy", "ask")
    if policy not in POLICIES:
        raise ValueError(f"policy must be one of {', '.join(POLICIES)}")
    parameters = raw.get("parameters")
    if not isinstance(parameters, dict) or parameters.get("type") != "object":
        raise ValueError("parameters must be an object schema")
    properties = parameters.get("properties", {})
    if not isinstance(properties, dict):
        raise ValueError("parameters.properties must be an object")
    for key, schema in properties.items():
        if not isinstance(schema, dict) or schema.get("type") not in ("string", "integer", "number", "boolean"):
            raise ValueError(f"parameter {key} must be a scalar schema")
    request = raw.get("request")
    if not isinstance(request, dict):
        raise ValueError("request must be an object")
    if str(request.get("method", "GET")).upper() not in ("GET", "POST"):
        raise ValueError("method must be GET or POST")
    url = str(request.get("url", ""))
    if not url.startswith(("http://", "https://")):
        raise ValueError("url must be http(s)")
    timeout_s = request.get("timeout_s", 10)
    if isinstance(timeout_s, bool) or not isinstance(timeout_s, int) or not (1 <= timeout_s <= 30):
        raise ValueError("timeout_s must be 1-30")
    headers = request.get("headers", {})
    secret_header = raw.get("secret_header", "authorization")
    if not isinstance(headers, dict) or not isinstance(secret_header, str) or not secret_header:
        raise ValueError("secret_header must be a non-empty string")
    params = set(properties)
    _templates(url, "request.url", params)
    _templates(request.get("query", {}), "request.query", params)
    _templates(headers, "request.headers", params)
    _templates(request.get("body", {}), "request.body", params)
    return raw


def _handler(spec: dict, client: httpx.AsyncClient, vault: Vault):
    name = spec["name"]
    request = spec["request"]
    method = str(request.get("method", "GET")).upper()
    url_template = str(request["url"])
    query = request.get("query", {}) or {}
    headers = request.get("headers", {}) or {}
    body = request.get("body", {}) or {}
    timeout_s = int(request.get("timeout_s", 10))
    properties = spec["parameters"].get("properties", {})
    required = spec["parameters"].get("required", [])
    secret_ref = spec.get("secret_ref")
    secret_header = spec.get("secret_header", "authorization")
    secret_prefix = spec.get("secret_prefix", "Bearer ")

    async def call(arguments: dict) -> ToolOutcome:
        unknown = set(arguments) - set(properties)
        if unknown:
            return ToolOutcome(False, f"error: unknown parameter(s): {', '.join(sorted(unknown))}")
        missing = [key for key in required if key not in arguments]
        if missing:
            return ToolOutcome(False, f"error: missing parameter(s): {', '.join(missing)}")
        values = {key: arguments[key] for key in arguments if key in properties}
        try:
            url = _fill(url_template, values)
            params = {key: _fill(str(value), values) for key, value in query.items()}
            header_values = {key: _fill(str(value), values) for key, value in headers.items()}
            payload = {key: _fill(value, values) if isinstance(value, str) else value for key, value in body.items()}
        except KeyError as exc:
            return ToolOutcome(False, f"error: plugin parameter {exc} was not supplied")
        if secret_ref:
            secret = vault.get(str(secret_ref))
            if secret is None:
                return ToolOutcome(False, f"error: secret '{secret_ref}' is not in the Credential Locker")
            header_values[secret_header] = f"{secret_prefix}{secret}"
        try:
            response = await client.request(
                method, url, params=params, headers=header_values,
                json=payload if method == "POST" else None, timeout=timeout_s,
            )
        except httpx.HTTPError as exc:
            return ToolOutcome(False, f"error: {name} request failed: {type(exc).__name__}")
        text = response.text
        if response.status_code >= 400:
            return ToolOutcome(False, f"error: HTTP {response.status_code} from {name}: {text[:300]}")
        if len(text.encode("utf-8")) > MAX_PLUGIN_RESPONSE:
            text = text.encode("utf-8")[:MAX_PLUGIN_RESPONSE].decode("utf-8", "replace") + "\n[truncated]"
        return ToolOutcome(True, text, summary=f"{name}({', '.join(sorted(values))})")

    return call


def load_plugins(directory: Path, client: httpx.AsyncClient, vault: Vault) -> tuple[list[ToolDef], list[PluginStatus]]:
    tools: list[ToolDef] = []
    statuses: list[PluginStatus] = []
    taken: set[str] = set()
    directory = Path(directory)
    if not directory.is_dir():
        return tools, statuses
    for path in sorted(directory.glob("*.json")):
        try:
            spec = _parse(path)
            name = spec["name"]
            if name in taken:
                raise ValueError(f"duplicate plugin name '{name}'")
            taken.add(name)
            tools.append(ToolDef(
                name=name,
                description=str(spec.get("description", "")),
                parameters=spec["parameters"],
                handler=_handler(spec, client, vault),
                risk=str(spec.get("risk", "network")),
                default_policy=str(spec.get("policy", "ask")),
                source=f"plugin:{path.name}",
                summary=lambda args, _name=name: f"{_name}({', '.join(sorted(args))})",
            ))
            statuses.append(PluginStatus(path.name, name, "loaded"))
        except Exception as exc:  # noqa: BLE001 - a bad file must not stop the engine
            statuses.append(PluginStatus(path.name, _safe_name(path), "skipped", str(exc)))
    return tools, statuses


def _safe_name(path: Path) -> str:
    try:
        raw = json.loads(path.read_text(encoding="utf-8"))
        return str(raw.get("name", path.stem)) if isinstance(raw, dict) else path.stem
    except Exception:
        return path.stem
```

```tsx
export function ToolsScreen({ client, connected }: { client: KiraClient; connected: boolean }) {

    if (!connected) return;

      .then(([tools, cps]) => { setRegistry(tools); setCheckpoints(cps.checkpoints); setError(null); })
      .catch((err) => setError(messageOf(err)));

  useEffect(() => { reload(); }, [reload]);

  async function changePolicy(name: string, policy: "allow" | "ask" | "deny" | null) {
    const secret = await ask({

      detail: "This changes what the agent may do without asking.",
      danger: policy === "allow",
    });
    if (!secret) return;

      await client.setToolPolicy(name, policy, secret);
      reload();
```

```tsx
import { useState } from "react";
import type { ApprovalChunk } from "../turn.ts";

const PREVIEW = 240;

export function ApprovalCard({ approval, onDecide }: { approval: ApprovalChunk; onDecide: (id: string, decision: "allow" | "deny", secret: string) => Promise<void> }): JSX.Element {
  const [secret, setSecret] = useState("");
  const [sent, setSent] = useState<"allow" | "deny" | null>(null);
  const [error, setError] = useState<string | null>(null);
  const preview = JSON.stringify(approval.arguments, null, 2);

  async function decide(decision: "allow" | "deny") {
    try {
      await onDecide(approval.id, decision, decision === "allow" ? secret : "");
      setSent(decision);
    } catch (err) {
      setError(err instanceof Error ? err.message : String(err));
    }
  }

  if (sent) return <p className="approval-note">{approval.tool}: {sent === "allow" ? "allowed" : "denied"}</p>;

  return (
    <div className="approval-card" role="group" aria-label={`approval for ${approval.tool}`}>
      <p className="work-meta">{approval.tool} · {approval.risk} · needs permission</p>
      <pre className="work-json">{preview.length > PREVIEW ? `${preview.slice(0, PREVIEW)}…` : preview}</pre>
      <input
        aria-label="approval secret"
        type="password"
        value={secret}
        onChange={(event) => setSecret(event.target.value)}
        placeholder="confirm secret"
        autoComplete="off"
      />
      <button type="button" onClick={() => decide("allow")} disabled={!secret.trim()}>allow</button>
      <button type="button" onClick={() => decide("deny")}>deny</button>
      {error && <p className="error">{error}</p>}
    </div>
  );
}
```
